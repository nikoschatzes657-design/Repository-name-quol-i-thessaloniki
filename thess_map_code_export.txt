from fastapi import APIRouter, Query, HTTPException
from sqlalchemy import text
from sqlalchemy.exc import SQLAlchemyError
from app.db.session import engine
from app.utils.geojson import rows_to_geojson
import logging
import time

logging.getLogger("sqlalchemy.engine").setLevel(logging.INFO)

router = APIRouter()

TABLE_WHITELIST = {
    "dk": "results.dk_indicators_v2",  # Δημοτικές Κοινότητες
}

@router.get("/health")
def health():
    return {"status": "ok"}

@router.get("/api/communities")
def communities(
    bbox: list[float] = Query(
        ...,
        min_items=4,
        max_items=4,
        description="minLon,minLat,maxLon,maxLat (EPSG:4326)"
    ),
    level: str = Query("dk", description="dk"),
    limit: int = 100
):
    t0 = time.perf_counter()

    minx, miny, maxx, maxy = bbox

    if minx >= maxx or miny >= maxy:
        raise HTTPException(400, "Invalid bbox")

    table = TABLE_WHITELIST.get(level.lower())

    if not table:
        raise HTTPException(400, "Invalid level. Use 'dk'.")

    sql = f"""
    SELECT
    to_jsonb(p) - 'geom' AS props,
    ST_AsGeoJSON(ST_Transform(p.geom, 4326))::json AS geometry
    FROM {table} p
    LIMIT :limit;
    """

    try:
        with engine.connect() as conn:
            res = conn.execute(
                text(sql),
                {
                    "minx": minx,
                    "miny": miny,
                    "maxx": maxx,
                    "maxy": maxy,
                    "limit": limit,
                }
            )

            cols = res.keys()
            rows = [dict(zip(cols, row)) for row in res]

        logging.info(
            "Returned %s features for level=%s in %.3f sec",
            len(rows),
            level,
            time.perf_counter() - t0
        )

        return rows_to_geojson(rows)

    except SQLAlchemyError as e:
        logging.exception("DB error")
        raise HTTPException(status_code=500, detail=str(e))

def rows_to_geojson(rows):
    feats = []
    for r in rows:
        geom = r.get("geometry")
        if not geom:
            continue
        props = r.get("props")
        if props is None:
            props = {k: v for k, v in r.items() if k != "geometry"}
        feats.append({"type": "Feature", "properties": props, "geometry": geom})
    return {"type": "FeatureCollection", "features": feats}

import { useMemo, useState } from "react";
import { computeScoreOT, computeScoreDK, clamp01 } from "src/lib/scoring.js";

function getName(props, level) {
  if (!props) return "Χωρίς όνομα";

  if (level === "dk") {
    return props.name || props.dk_name || props.community || "Χωρίς όνομα";
  }

  return (
    props.fua_name ||
    props.mun_name ||
    props.community ||
    props.name ||
    `OT ${props.id ?? ""}` ||
    "Χωρίς όνομα"
  );
}

function getScore(props, level, weights) {
  const raw =
    level === "dk"
      ? computeScoreDK(props, weights)
      : computeScoreOT(props, weights);

  return clamp01(raw);
}

function getClassLabel(score) {
  if (!Number.isFinite(score)) return "Μη διαθέσιμη";

  if (score < 0.2) return "Πολύ χαμηλή";
  if (score < 0.4) return "Χαμηλή";
  if (score < 0.6) return "Μέτρια";
  if (score < 0.8) return "Υψηλή";
  return "Πολύ υψηλή";
}

export default function RankingPanel({
  geojson,
  allGeojson,
  weights,
  level = "dk",
}) {
  const [scope, setScope] = useState("all");
  const [count, setCount] = useState(5);

  const sourceFeatures = useMemo(() => {
    if (scope === "viewport") {
      return geojson?.features ? [...geojson.features] : [];
    }

    return allGeojson?.features ? [...allGeojson.features] : [];
  }, [scope, geojson, allGeojson]);

  const ranking = useMemo(() => {
    return sourceFeatures
      .map((feature) => {
        const props = feature.properties || {};
        const score = getScore(props, level, weights);

        return {
          id: props.id ?? props.objectid ?? props.fid ?? getName(props, level),
          name: getName(props, level),
          score,
          category: getClassLabel(score),
          props,
        };
      })
      .filter((row) => Number.isFinite(row.score))
      .sort((a, b) => b.score - a.score);
  }, [sourceFeatures, level, weights]);

  const topRows = ranking.slice(0, count);
  const bottomRows = [...ranking].reverse().slice(0, count);

  const title =
    level === "dk"
      ? "Κατάταξη Δημοτικών Κοινοτήτων"
      : "Κατάταξη Οικοδομικών Τετραγώνων";

  const scoreLabel =
    level === "dk"
      ? "QoL-i"
      : "Δείκτης προσβασιμότητας";

  return (
    <>
      <div className="panel-title">{title}</div>

      <div className="dist-controls">
        <label htmlFor="ranking-scope">Πεδίο:</label>
        <select
          id="ranking-scope"
          value={scope}
          onChange={(e) => setScope(e.target.value)}
        >
          <option value="all">Όλες οι περιοχές</option>
          <option value="viewport">Τρέχουσα προβολή χάρτη</option>
        </select>

        <label htmlFor="ranking-count">Πλήθος:</label>
        <select
          id="ranking-count"
          value={count}
          onChange={(e) => setCount(Number(e.target.value))}
        >
          <option value={5}>Top/Bottom 5</option>
          <option value={10}>Top/Bottom 10</option>
          <option value={15}>Top/Bottom 15</option>
        </select>

        <span className="badge">N={ranking.length}</span>
      </div>

      <div className="ranking-grid">
        <RankingTable

          rows={topRows}
          scoreLabel={scoreLabel}
        />

        <RankingTable

        />
      </div>

function setWeightConstrained(prev, id, newVal) {
  const keys = Object.keys(prev);
  const clamped = Math.max(0, Math.min(1, Number(newVal) || 0));
  const others = keys.filter((k) => k !== id);
  const sumOthers = others.reduce((a, k) => a + Number(prev[k] || 0), 0);

  if (clamped >= 1 || sumOthers === 0) {
    return Object.fromEntries(keys.map((k) => [k, k === id ? 1 : 0]));
  }

  const remaining = 1 - clamped;
  const ratio = remaining / sumOthers;
  const next = {};
  for (const k of keys) {
    next[k] = k === id ? clamped : Number(prev[k] || 0) * ratio;
  }
  return next;
}

export default function WeightsPanel({
  level = "ot",
  weights,
  setWeights,
  lockSumToOne,
  setLockSumToOne,
  showDynamic,
  setShowDynamic,
}) {
  const { groups, flat, defaults, profiles } = useMemo(() => {
    if (level === "dk") {
      return {
        groups: DK_GROUPS,
        flat: DK_INDICATORS,
        defaults: DEFAULT_WEIGHTS_DK,
        profiles: DK_WEIGHT_PROFILES,
      };
    }
    return {
      groups: GROUPS_OT,
      flat: FLAT_OT,
      defaults: DEFAULT_WEIGHTS_OT,
      profiles: null,
    };
  }, [level]);

  const [open, setOpen] = useState(() =>
    Object.fromEntries(groups.map((g) => [g.title, true]))
  );

  const [selectedProfile, setSelectedProfile] = useState(
    level === "dk" ? "thesis" : "custom"
  );

  const toggle = (title) =>
    setOpen((o) => ({
      ...o,
      [title]: !o[title],
    }));

  const applyProfile = (id) => {
    setSelectedProfile(id);
    if (!profiles) return;
    const prof = profiles.find((p) => p.id === id);
    if (prof && prof.weights) {
      setWeights({ ...prof.weights });
    }
  };

  const setW = (id, v) => {
    const numeric = Number(v);
    const clean = Number.isFinite(numeric) ? numeric : 0;
    setWeights((prev) =>
      lockSumToOne
        ? setWeightConstrained(prev, id, clean)
        : {
            ...prev,
            [id]: Math.max(0, Math.min(1, clean)),
          }
    );
    if (level === "dk" && selectedProfile !== "custom") {
      setSelectedProfile("custom");
    }
  };

  const reset = () => {
    setWeights(defaults);
    if (level === "dk") setSelectedProfile("thesis");
  };

  const zeroAll = () =>
    setWeights(Object.fromEntries(flat.map((it) => [it.id, 0])));

  const sum = Object.values(weights || {}).reduce(
    (a, b) => a + Number(b || 0),
    0
  );

  const groupSums = useMemo(() => {
    const res = {};
    groups.forEach((g) => {
      res[g.title] = (g.items || []).reduce(
        (acc, it) => acc + Number(weights?.[it.id] || 0),
        0
      );
    });
    return res;
  }, [groups, weights]);

  return (
    <div className="side-panel">
      <div style={{ display: "flex", alignItems: "center", gap: 8 }}>
        <input
          id="toggle-dynamic"
          type="checkbox"
          checked={showDynamic}
          onChange={(e) => setShowDynamic(e.target.checked)}
        />
        <label
          htmlFor="toggle-dynamic"
          style={{ userSelect: "none", fontWeight: 500 }}
        >
          Εμφάνιση θεματικών πολυγώνων
        </label>
      </div>

      <div className="lock-row" style={{ marginTop: 8, display: "flex", gap: 8 }}>
        <div style={{ flex: 1 }}>
          <input
            id="lock-sum"
            type="checkbox"
            checked={lockSumToOne}
            onChange={(e) => setLockSumToOne(e.target.checked)}
          />
          <label htmlFor="lock-sum">Κλείδωμα: άθροισμα βαρών = 1</label>
        </div>

        {level === "dk" && profiles && (
          <div style={{ flex: 1 }}>
            <div style={{ fontSize: 12, opacity: 0.8 }}>Προφίλ βαρών (DK)</div>
            <select

              <div className="group-body">
                {g.items.map((it) => {
                  const value = Number(

import { useCallback, useEffect, useRef, useState } from "react";
import { computeScoreOT, computeScoreDK, clamp01 } from "src/lib/scoring.js";
import { DEFAULT_WEIGHTS_OT } from "src/lib/constants.js";

/* Εμπλουτισμός κάθε feature με _score και _score_vis */
function enrich(geojson, level, weights) {
  const data = JSON.parse(JSON.stringify(geojson || { type: "FeatureCollection", features: [] }));
  for (const f of data.features || []) {
    const p = f.properties || {};
    const raw = level === "dk"
      ? computeScoreDK(p, weights)
      : computeScoreOT(p, weights || DEFAULT_WEIGHTS_OT);
    p._score = raw;
    p._score_vis = clamp01(raw) ?? 0;
  }
  return data;
}

export function useMapCommunities({ apiBase, level, weights }) {
  const [geojson, setGeojson] = useState(null);
  const [allGeojson, setAllGeojson] = useState(null);
  const [loading, setLoading] = useState(false);

  const viewportAbortRef = useRef(null);
  const API = apiBase || import.meta.env.VITE_API_URL || "http://127.0.0.1:8000";

  const fetchByBbox = useCallback(
    async (bbox) => {
      try {
        viewportAbortRef.current?.abort?.();
        const ctrl = new AbortController();
        viewportAbortRef.current = ctrl;

        const url = new URL(`${API}/api/communities`);
        for (const v of bbox) url.searchParams.append("bbox", String(v));
        url.searchParams.set("level", level);

        setLoading(true);
        const resp = await fetch(url.toString(), { signal: ctrl.signal });
        if (!resp.ok) throw new Error(`API ${resp.status}`);
        const data = await resp.json();
        const enriched = enrich(data, level, weights);
        console.log("FETCHED", level, enriched.features?.length, enriched.features?.[0]?.properties);
        setGeojson(enriched);
        return enriched;
      } finally {
        setLoading(false);
      }
    },
    [API, level, weights]
  );

  const fetchAll = useCallback(
  async () => {
    const url = new URL(`${API}/api/communities`);

    // Bbox Θεσσαλονίκης
    [22.80, 40.50, 23.10, 40.80].forEach(v =>
      url.searchParams.append("bbox", String(v))
    );

    url.searchParams.set("level", level);

    const resp = await fetch(url.toString());

    if (!resp.ok) {
      const empty = { type: "FeatureCollection", features: [] };
      setAllGeojson(empty);
      setGeojson(empty);
      return null;
    }

    const data = await resp.json();
    const enriched = enrich(data, level, weights);

    // ΠΟΛΥ ΣΗΜΑΝΤΙΚΟ:
    // Για DK-only, το ίδιο dataset μπαίνει και στο χάρτη και στα panels.
    setAllGeojson(enriched);
    setGeojson(enriched);

    console.log(
      "FETCH ALL DK:",
      enriched.features?.length ?? 0,
      "features"
    );

    return enriched;
  },
  [API, level, weights]
);

  useEffect(() => { fetchAll(); }, [fetchAll]);

  // Re-enrich “all” όταν αλλάζουν βάρη στο τρέχον level
  useEffect(() => {
  if (!allGeojson?.features) return;

  const enriched = enrich(allGeojson, level, weights);

  setAllGeojson(enriched);
  setGeojson(enriched);
}, [weights, level]);
  useEffect(() => () => viewportAbortRef.current?.abort?.(), []);

  return { geojson, allGeojson, setGeojson, fetchByBbox, loading };
}

    RDYLGN.length - 1

export function getQoLClass(value) {
  const v = Number(value);

  if (!Number.isFinite(v)) {
    return {
      label: "Μη διαθέσιμη",
      range: "–",
      index: -1,
    };
  }

  const clamped = Math.max(0, Math.min(1, v));

  const idx = QOL_CLASSES.findIndex((c, i) => {
    const isLast = i === QOL_CLASSES.length - 1;
    return isLast
      ? clamped >= c.min && clamped <= c.max
      : clamped >= c.min && clamped < c.max;
  });

  const klass = QOL_CLASSES[idx] ?? QOL_CLASSES[0];

  return {
    label: klass.label,
    range: `${klass.min.toFixed(2)}–${klass.max.toFixed(2)}`,
    index: idx,
  };
}

// src/lib/scoring.js
import * as C from "./constants.js";

/* pick από πολλαπλά keys (μόνο για OT) */
export function pickValue(props, keys) {
  if (!props) return null;
  for (const k of (keys || [])) {
    const v = props[k];
    if (v !== undefined && v !== null && v !== "") return Number(v);
  }
  return null;
}

export function clamp01(x) {
  return Number.isFinite(x) ? Math.max(0, Math.min(1, Number(x))) : null;
}

/* ====== OT (flat proximity) ====== */
export function computeScoreOT(props, weights) {
  try {
    const items = Array.isArray(C.FLAT_OT) ? C.FLAT_OT : [];
    let ssum = 0, wsum = 0;
    for (const it of items) {
      const w = Number(weights?.[it.id] ?? 0);
      if (w <= 0) continue;
      const v = pickValue(props, it.keys);
      if (Number.isFinite(v)) { ssum += w * v; wsum += w; }
    }
    return wsum > 0 ? clamp01(ssum / wsum) : null;
  } catch {
    return null;
  }
}

/* ====== DK (flat sliders, με inversion όπου χρειάζεται) ====== */
export function computeScoreDK(props, weights) {
  try {
    // Αν για κάποιο λόγο δεν υπάρχει DK_INDICATORS, μην πεθάνουμε.
    const metas = Array.isArray(C.DK_INDICATORS) ? C.DK_INDICATORS : [];

    // Αν ούτε αυτά υπάρχουν, έστω βάδισε πάνω στα keys των weights (χωρίς invert).
    const safeMetas = metas.length
      ? metas
      : Object.keys(weights || {}).map(id => ({ id, invert: false }));

    let ssum = 0, wsum = 0;

    for (const it of safeMetas) {
      const w = Number(weights?.[it.id] ?? 0);
      if (w <= 0) continue;

      let v = props?.[it.id];
      if (v === null || v === undefined || v === "") continue;

      v = Number(v);
      if (!Number.isFinite(v)) continue;

      // clamp στο [0,1]
      v = Math.max(0, Math.min(1, v));

      // ρητό invert μόνο όταν ζητείται
      if (it.invert === true) v = 1 - v;

      ssum += w * v;
      wsum += w;
    }

    return wsum > 0 ? Number((ssum / wsum).toFixed(6)) : null;
  } catch {
    // Τελευταία άμυνα: ποτέ άσπρη οθόνη.
    return null;
  }
}

// Δώσε k χρώματα από την παλέτα (ακόμη κι αν είναι μεγαλύτερη/μικρότερη).
export function samplePalette(base, k) {
  const arr = Array.isArray(base) ? base : PALETTES.Viridis;
  const n = Math.max(3, Math.min(9, Number(k) || 6));
  if (arr.length === n) return [...arr];
  const out = [];
  for (let i = 0; i < n; i++) {
    const t = i / (n - 1); // 0..1
    const idx = Math.round(t * (arr.length - 1));
    out.push(arr[idx]);
  }
  return out;
}

const API = import.meta.env.VITE_API_URL || "http://127.0.0.1:8000";

export async function fetchCommunitiesByBbox(bounds, level = "ot") {
  const url = `${API}/api/communities?level=${encodeURIComponent(level)}&bbox=${bounds.getWest()}&bbox=${bounds.getSouth()}&bbox=${bounds.getEast()}&bbox=${bounds.getNorth()}`;
  const resp = await fetch(url);
  if (!resp.ok) throw new Error(`API ${resp.status}`);
  return resp.json();
}

export async function fetchAllCommunities(level = "ot") {
  const url = `${API}/api/communities?level=${encodeURIComponent(level)}&bbox=-180&bbox=-90&bbox=180&bbox=90`;
  const resp = await fetch(url);
  if (!resp.ok) throw new Error(`API ${resp.status}`);
  return resp.json();
}
