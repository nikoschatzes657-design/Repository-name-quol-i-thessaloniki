from fastapi import APIRouter, Query, HTTPException
from sqlalchemy import text
from sqlalchemy.exc import SQLAlchemyError
from app.db.session import engine

import logging
logging.getLogger("sqlalchemy.engine").setLevel(logging.INFO)  # verbose SQL

router = APIRouter()

TABLE_WHITELIST = {
    "ot": "public.polygons_with_means",     # ΞΞΉΞΊΞΏΞ΄ΞΏΞΌΞΉΞΊΞ¬ Ξ¤ΞµΟ„ΟΞ¬Ξ³Ο‰Ξ½Ξ±
    "dk": "public.dk_indicators_v2",        # Ξ”Ξ·ΞΌΞΏΟ„ΞΉΞΊΞ­Ο‚ ΞΞΏΞΉΞ½ΟΟ„Ξ·Ο„ΞµΟ‚ (matview)

@router.get("/health")
def health():
    return {"status": "ok"}

@router.get("/api/communities")
def communities(
    bbox: list[float] = Query(..., min_items=4, max_items=4, description="minLon,minLat,maxLon,maxLat (EPSG:4326)"),
    level: str = Query("ot", description="ot | dk"),
    limit: int = 20000
):
    minx, miny, maxx, maxy = bbox
    if minx >= maxx or miny >= maxy:
        raise HTTPException(400, "Invalid bbox")

    table = TABLE_WHITELIST.get(level.lower())
    if not table:
        raise HTTPException(400, "Invalid level. Use 'ot' or 'dk'.")

    sql = f"""
    WITH b AS (
      SELECT ST_Transform(ST_MakeEnvelope(:minx,:miny,:maxx,:maxy,4326), 3857) AS env
    )
    SELECT
      to_jsonb(p) - 'geom' AS props,
      ST_AsGeoJSON(
        ST_Transform(
          CASE WHEN ST_SRID(p.geom)=0 THEN ST_SetSRID(p.geom,2100) ELSE p.geom END,
          4326
        )
      )::json AS geometry
    FROM {table} p, b
    WHERE ST_Intersects(
      ST_Transform(CASE WHEN ST_SRID(p.geom)=0 THEN ST_SetSRID(p.geom,2100) ELSE p.geom END, 3857),
      b.env
    )
    LIMIT :limit;
    """

    try:
        with engine.connect() as conn:
            res = conn.execute(
                text(sql),
                {"minx": minx, "miny": miny, "maxx": maxx, "maxy": maxy, "limit": limit}
            )
            cols = res.keys()
            rows = [dict(zip(cols, row)) for row in res]
        return rows_to_geojson(rows)
    except SQLAlchemyError as e:
        logging.exception("DB error")
        raise HTTPException(status_code=500, detail=str(e))

def rows_to_geojson(rows):
    feats = []
    for r in rows:
        geom = r.get("geometry")
        if not geom:
            continue
        props = r.get("props")
        if props is None:
            props = {k: v for k, v in r.items() if k != "geometry"}
        feats.append({"type": "Feature", "properties": props, "geometry": geom})
    return {"type": "FeatureCollection", "features": feats}

export default function DistPanel({ geojson, allGeojson, weights, selectedBin, onSelectBin, level="ot" }) {
  const wrapRef = useRef(null);
  const canvasRef = useRef(null);
  const chartRef = useRef(null);

  const [metric, setMetric] = useState("score");
  const [scope, setScope] = useState("all");

  const indicators = useMemo(() => {
    return level === "dk"
      ? [{ id: "score", label: "Ξ£Ο…Ξ½ΞΏΞ»ΞΉΞΊΟ Score (ΟƒΟ„Ξ±ΞΈΞΌΞΉΟƒΞΌΞ­Ξ½ΞΏ DK)" }, ...DK_INDICATORS]
      : [{ id: "score", label: "Ξ£Ο…Ξ½ΞΏΞ»ΞΉΞΊΟ Score (ΞΌΞµ Ξ²Ξ¬ΟΞ·)" }, ...FLAT_OT.map(i => ({ id: i.id, label: i.label }))];
  }, [level]);

  const pickByMetric = (props) => {
    if (!props) return null;
    let val = null;
    if (metric === "score") {
      val = level === "dk" ? computeScoreDK(props, weights) : computeScoreOT(props, weights);
    } else {
      if (level === "dk") val = Number(props[metric]);
      else {
        const meta = FLAT_OT.find(x => x.id === metric);
        if (meta) val = pickValue(props, meta.keys);
      }
    }
    return Number.isFinite(val) ? Math.max(0, Math.min(1, Number(val))) : null;
  };

  const sourceFeatures = useMemo(() => {
    if (scope === "all") return allGeojson?.features ? [...allGeojson.features] : [];
    return geojson?.features ? [...geojson.features] : [];
  }, [scope, allGeojson, geojson]);

  const values = useMemo(() => {
    if (!sourceFeatures.length) return [];
    return sourceFeatures
      .map(f => pickByMetric(f.properties))
      .filter(v => Number.isFinite(v));
  }, [sourceFeatures, metric, weights, level]);

  const domain = useMemo(() => {
    if (!values.length) return [0, 1];
    const min = Math.min(...values);
    const max = Math.max(...values);
    if (min >= 0 && max <= 1.0001) return [0, 1];
    return [min, max];
  }, [values]);

  const BINS = 10;
  const EDGES = useMemo(() => {
    const [min, max] = domain;
    const step = (max - min) / BINS;
    return Array.from({ length: BINS + 1 }, (_, i) => min + i * step);
  }, [domain]);
  const CENTERS = useMemo(() => Array.from({ length: BINS }, (_, i) => (EDGES[i] + EDGES[i + 1]) / 2), [EDGES]);

    const [min, max] = domain;
    const step = (max - min) / BINS;
    const counts = new Array(BINS).fill(0);
    for (const v of vals) {
      let idx = Math.floor((v - min) / step);
      if (idx < 0) idx = 0;
      if (idx > BINS - 1) idx = BINS - 1;
      counts[idx]++;
    }
    const total = Math.max(1, vals.length);
    const cdf = counts.map((_, i) => counts.slice(0, i + 1).reduce((a,b)=>a+b,0) / total);
    return { counts, cdf };
  };

  useEffect(() => {
    if (!canvasRef.current) return;

    const ctx = canvasRef.current.getContext("2d");
    if (!ctx) return;

    const { counts, cdf } = makeHistogramCDF(values);
    const barData = CENTERS.map((x, i) => ({ x, y: counts[i] }));
    const cdfData = CENTERS.map((x, i) => ({ x, y: cdf[i] }));

    const barColors = barData.map(d => colorFor((d.x - domain[0]) / (domain[1] - domain[0])));

        ],
      },
      options: {

        animation: false,

        onClick: (evt) => {

          if (points?.length) {
            const i = points[0].index;
            onSelectBin(prev => (prev === i ? null : i));
          }
        },
        scales: {

          y: { beginAtZero: true, title: { display: true, text: "Ξ Ξ»Ξ®ΞΈΞΏΟ‚ Ο€ΞΏΞ»Ο…Ξ³ΟΞ½Ο‰Ξ½ (N)" } },

function setWeightConstrained(prev, id, newVal) {
  const keys = Object.keys(prev);
  const clamped = Math.max(0, Math.min(1, Number(newVal) || 0));
  const others = keys.filter(k => k !== id);
  const sumOthers = others.reduce((a, k) => a + Number(prev[k] || 0), 0);
  if (clamped >= 1 || sumOthers === 0) {
    return Object.fromEntries(keys.map(k => [k, k === id ? 1 : 0]));
  }
  const R = 1 - clamped;
  const ratio = R / sumOthers;
  const next = {};
  for (const k of keys) next[k] = (k === id) ? clamped : Number(prev[k] || 0) * ratio;
  return next;
}

function enrich(geojson, level, weights) {
  const data = JSON.parse(JSON.stringify(geojson || { type: "FeatureCollection", features: [] }));
  for (const f of data.features || []) {
    const p = f.properties || {};
    const raw = level === "dk"
      ? computeScoreDK(p, weights)
      : computeScoreOT(p, weights || DEFAULT_WEIGHTS_OT);
    p._score = raw;
    p._score_vis = clamp01(raw) ?? 0;
  }
  return data;
}

// src/lib/scoring.js
import * as C from "./constants.js";

/* pick Ξ±Ο€Ο Ο€ΞΏΞ»Ξ»Ξ±Ο€Ξ»Ξ¬ keys (ΞΌΟΞ½ΞΏ Ξ³ΞΉΞ± OT) */
export function pickValue(props, keys) {
  if (!props) return null;
  for (const k of (keys || [])) {
    const v = props[k];
    if (v !== undefined && v !== null && v !== "") return Number(v);
  }
  return null;
}

export function clamp01(x) {
  return Number.isFinite(x) ? Math.max(0, Math.min(1, Number(x))) : null;
}

/* ====== OT (flat proximity) ====== */
export function computeScoreOT(props, weights) {
  try {
    const items = Array.isArray(C.FLAT_OT) ? C.FLAT_OT : [];
    let ssum = 0, wsum = 0;
    for (const it of items) {
      const w = Number(weights?.[it.id] ?? 0);
      if (w <= 0) continue;
      const v = pickValue(props, it.keys);
      if (Number.isFinite(v)) { ssum += w * v; wsum += w; }
    }
    return wsum > 0 ? clamp01(ssum / wsum) : null;
  } catch {
    return null;
  }
}

/* ====== DK (flat sliders, ΞΌΞµ inversion ΟΟ€ΞΏΟ… Ο‡ΟΞµΞΉΞ¬Ξ¶ΞµΟ„Ξ±ΞΉ) ====== */
export function computeScoreDK(props, weights) {
  try {
    // Ξ‘Ξ½ Ξ³ΞΉΞ± ΞΊΞ¬Ο€ΞΏΞΉΞΏ Ξ»ΟΞ³ΞΏ Ξ΄ΞµΞ½ Ο…Ο€Ξ¬ΟΟ‡ΞµΞΉ DK_INDICATORS, ΞΌΞ·Ξ½ Ο€ΞµΞΈΞ¬Ξ½ΞΏΟ…ΞΌΞµ.
    const metas = Array.isArray(C.DK_INDICATORS) ? C.DK_INDICATORS : [];

    // Ξ‘Ξ½ ΞΏΟΟ„Ξµ Ξ±Ο…Ο„Ξ¬ Ο…Ο€Ξ¬ΟΟ‡ΞΏΟ…Ξ½, Ξ­ΟƒΟ„Ο‰ Ξ²Ξ¬Ξ΄ΞΉΟƒΞµ Ο€Ξ¬Ξ½Ο‰ ΟƒΟ„Ξ± keys Ο„Ο‰Ξ½ weights (Ο‡Ο‰ΟΞ―Ο‚ invert).
    const safeMetas = metas.length
      ? metas
      : Object.keys(weights || {}).map(id => ({ id, invert: false }));

    let ssum = 0, wsum = 0;

    for (const it of safeMetas) {
      const w = Number(weights?.[it.id] ?? 0);
      if (w <= 0) continue;

      let v = props?.[it.id];
      if (v === null || v === undefined || v === "") continue;

      v = Number(v);
      if (!Number.isFinite(v)) continue;

      // clamp ΟƒΟ„ΞΏ [0,1]
      v = Math.max(0, Math.min(1, v));

      // ΟΞ·Ο„Ο invert ΞΌΟΞ½ΞΏ ΟΟ„Ξ±Ξ½ Ξ¶Ξ·Ο„ΞµΞ―Ο„Ξ±ΞΉ
      if (it.invert === true) v = 1 - v;

      ssum += w * v;
      wsum += w;
    }

    return wsum > 0 ? Number((ssum / wsum).toFixed(6)) : null;
  } catch {
    // Ξ¤ΞµΞ»ΞµΟ…Ο„Ξ±Ξ―Ξ± Ξ¬ΞΌΟ…Ξ½Ξ±: Ο€ΞΏΟ„Ξ­ Ξ¬ΟƒΟ€ΟΞ· ΞΏΞΈΟΞ½Ξ·.
    return null;
  }
}

# React + Vite

This template provides a minimal setup to get React working in Vite with HMR and some ESLint rules.

Currently, two official plugins are available:

- [@vitejs/plugin-react](https://github.com/vitejs/vite-plugin-react/blob/main/packages/plugin-react) uses [Babel](https://babeljs.io/) for Fast Refresh
- [@vitejs/plugin-react-swc](https://github.com/vitejs/vite-plugin-react/blob/main/packages/plugin-react-swc) uses [SWC](https://swc.rs/) for Fast Refresh
